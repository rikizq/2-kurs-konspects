```C#
using System;

class Program
{
    static void Main()
    {
        Console.Write("Введите размер ромба: ");
        int n = int.Parse(Console.ReadLine());

        int i = 1;

        while (i <= n)
        {
            int spaces = Math.Abs(n / 2 - i);
            int stars = n - spaces * 2;

            int j = 0;

            while (j < spaces)
            {
                Console.Write(" ");
                j++;
            }

            j = 0;

            while (j < stars)
            {
                Console.Write("*");
                j++;
            }

            Console.WriteLine();
            i++;
        }
    }
}
```
