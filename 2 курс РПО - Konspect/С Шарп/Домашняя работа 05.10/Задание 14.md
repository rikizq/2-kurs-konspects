using System;

class Program
{
    static void Main()
    {
        int p = -1;

        while (p != 0)
        {
            Console.WriteLine();
            Console.WriteLine("1 - Посчитать строки");
            Console.WriteLine("2 - Посчитать опоздания");
            Console.WriteLine("3 - Посчитать зарплату");
            Console.WriteLine("0 - Выход");
            Console.Write("Выберите действие: ");
            p = int.Parse(Console.ReadLine());

            if (p == 1)
            {
                Console.Write("Желаемый доход: ");
                int a = int.Parse(Console.ReadLine());

                Console.Write("Количество опозданий: ");
                int b = int.Parse(Console.ReadLine());

                int c = 0;
                int t = b;
                while (t >= 3)
                {
                    c += 20;
                    t -= 3;
                }

                int d = a + c;
                int x = 0;
                int m = 0;
                while (m < d)
                {
                    m += 50;
                    x += 100;
                }

                Console.WriteLine("Нужно написать строк: " + x);
            }
            else if (p == 2)
            {
                Console.Write("Количество строк: ");
                int a = int.Parse(Console.ReadLine());

                Console.Write("Желаемая зарплата: ");
                int b = int.Parse(Console.ReadLine());

                int c = 0;
                int t = a;
                while (t >= 100)
                {
                    c += 50;
                    t -= 100;
                }

                if (c < b)
                {
                    Console.WriteLine("Не сможет получить эту сумму.");
                }
                else
                {
                    int k = 0;
                    while (c - 20 >= b)
                    {
                        k += 3;
                        c -= 20;
                    }
                    Console.WriteLine("Может опоздать максимум: " + (k + 2));
                }
            }
            else if (p == 3)
            {
                Console.Write("Количество строк: ");
                int a = int.Parse(Console.ReadLine());

                Console.Write("Количество опозданий: ");
                int b = int.Parse(Console.ReadLine());

                int c = 0;
                int d = 0;
                int t = a;
                while (t >= 100)
                {
                    c += 50;
                    t -= 100;
                }

                t = b;
                while (t >= 3)
                {
                    d += 20;
                    t -= 3;
                }

                int s = c - d;
                if (s < 0)
                {
                    s = 0;
                }
                Console.WriteLine("Заплатят: " + s + "$");
            }
            else if (p != 0)
            {
                Console.WriteLine("Неверный пункт меню.");
            }
        }
    }
}
