```C#
using System;

class Program
{
    static void Main()
    {
        Console.Write("Введите x: ");
        int x = int.Parse(Console.ReadLine());

        Console.Write("Введите y: ");
        int y = int.Parse(Console.ReadLine());

        int result = 1;
        int i = 0;

        while (i < y)
        {
            result = result * x;
            i++;
        }

        Console.WriteLine("Результат: " + result);
    }
}
```
