```C#
using System;

class Program
{
    static void Main()
    {
        Console.Write("Введите k: ");
        int k = int.Parse(Console.ReadLine());

        int i = 1;

        while (i <= 10)
        {
            Console.WriteLine(k + " x " + i + " = " + k * i);
            i++;
        }
    }
}
```
