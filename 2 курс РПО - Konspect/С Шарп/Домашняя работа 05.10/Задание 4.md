```C#
using System;

class Program
{
    static void Main()
    {
        int number = 1;
        int sum = 0;

        while (number != 0)
        {
            Console.Write("Введите число: ");
            number = int.Parse(Console.ReadLine());

            sum += number;
        }

        Console.WriteLine("Сумма: " + sum);
    }
}
```
