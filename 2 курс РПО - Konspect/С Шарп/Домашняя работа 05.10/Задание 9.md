```C#
using System;

class Program
{
    static void Main()
    {
        int i = 1;
        int sum = 0;

        while (i <= 1000)
        {
            sum += i;
            i++;
        }

        double average = (double)sum / 1000;

        Console.WriteLine("Среднее арифметическое: " + average);
    }
}
```
