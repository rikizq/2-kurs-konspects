```C#
using System;

class Program
{
    static void Main()
    {
        Random random = new Random();

        int secret = random.Next(1, 501);
        int number = -1;
        int attempts = 0;

        DateTime start = DateTime.Now;

        Console.WriteLine("Я загадал число от 1 до 500.");
        Console.WriteLine("Для выхода введите 0.");

        while (number != secret && number != 0)
        {
            Console.Write("Введите число: ");
            number = int.Parse(Console.ReadLine());

            if (number == 0)
                Console.WriteLine("Вы вышли из игры.");
            else if (number < secret)
            {
                Console.WriteLine("Загаданное число больше.");
                attempts++;
            }
            else if (number > secret)
            {
                Console.WriteLine("Загаданное число меньше.");
                attempts++;
            }
            else
            {
                attempts++;

                DateTime finish = DateTime.Now;
                TimeSpan time = finish - start;

                Console.WriteLine("Вы угадали!");
                Console.WriteLine("Попыток: " + attempts);
                Console.WriteLine("Время: " + time.TotalSeconds + " секунд");
            }
        }
    }
}
```
