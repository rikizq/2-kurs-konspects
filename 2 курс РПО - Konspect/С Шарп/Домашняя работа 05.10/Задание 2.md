```C#
using System;

class Program
{
    static void Main()
    {
        Console.Write("Введите первую границу: ");
        int a = int.Parse(Console.ReadLine());

        Console.Write("Введите вторую границу: ");
        int b = int.Parse(Console.ReadLine());

        if (a > b)
        {
            int x = a;
            a = b;
            b = x;
        }

        int i = a;

        Console.WriteLine("Все числа:");

        while (i <= b)
        {
            Console.Write(i + " ");
            i++;
        }

        Console.WriteLine();
        Console.WriteLine("Чётные:");
        i = a;

        while (i <= b)
        {
            if (i % 2 == 0)
                Console.Write(i + " ");
            i++;
        }

        Console.WriteLine();
        Console.WriteLine("Нечётные:");

        i = a;

        while (i <= b)
        {
            if (i % 2 != 0)
                Console.Write(i + " ");
            i++;
        }

        Console.WriteLine();
        Console.WriteLine("Кратные 7:");
        i = a;

        while (i <= b)
        {
            if (i % 7 == 0)
                Console.Write(i + " ");
            i++;
        }
    }
}
```
