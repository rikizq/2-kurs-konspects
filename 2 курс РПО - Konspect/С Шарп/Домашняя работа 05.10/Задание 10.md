```C#
using System;

class Program
{
    static void Main()
    {
        Console.Write("Введите a: ");
        int a = int.Parse(Console.ReadLine());

        long result = 1;

        while (a <= 20)
        {
            result = result * a;
            a++;
        }

        Console.WriteLine("Произведение: " + result);
    }
}
```
