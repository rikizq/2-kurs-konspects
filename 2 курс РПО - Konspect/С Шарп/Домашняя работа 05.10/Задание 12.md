```C#
using System;

class Program
{
    static void Main()
    {
        Random random = new Random();

        Console.WriteLine("1 - Лёгкий уровень");
        Console.WriteLine("2 - Средний уровень");
        Console.WriteLine("3 - Сложный уровень");

        Console.Write("Выберите уровень: ");
        int level = int.Parse(Console.ReadLine());

        int questions = 5;
        int max = 5;

        if (level == 2)
        {
            questions = 10;
            max = 10;
        }
        else if (level == 3)
        {
            questions = 15;
            max = 20;
        }

        int correct = 0;
        int i = 0;

        while (i < questions)
        {
            int a = random.Next(1, max + 1);
            int b = random.Next(1, max + 1);

            Console.Write(a + " x " + b + " = ");
            int answer = int.Parse(Console.ReadLine());

            if (answer == a * b)
            {
                Console.WriteLine("Правильно!");
                correct++;
            }
            else
                Console.WriteLine("Неправильно. Ответ: " + a * b);

            i++;
        }

        Console.WriteLine("Правильных ответов: " + correct);

        if (correct >= questions * 0.9)
            Console.WriteLine("Оценка: 5");
        else if (correct >= questions * 0.7)
            Console.WriteLine("Оценка: 4");
        else if (correct >= questions * 0.5)
            Console.WriteLine("Оценка: 3");
        else
            Console.WriteLine("Оценка: 2");
    }
}
```
