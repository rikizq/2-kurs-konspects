```C#
using System;

class Program
{
    static void Main()
    {
        Console.Write("Введите a: ");
        int a = int.Parse(Console.ReadLine());

        int sum = 0;

        while (a <= 500)
        {
            sum += a;
            a++;
        }

        Console.WriteLine("Сумма: " + sum);
    }
}
```
