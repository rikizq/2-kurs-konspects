```C#
using System;

class Product
{
    public string Name { get; set; }
    public decimal Price { get; set; }
    public int Quantity { get; set; }

    public Product(string name, decimal price, int quantity)
    {
        Name = name;
        Price = price;
        Quantity = quantity;
    }

    public decimal GetTotalValue()
    {
        return Price * Quantity;
    }

    public bool Sell(int amount)
    {
        if (amount > 0 && amount <= Quantity)
        {
            Quantity = Quantity - amount;
            return true;
        }
        else
        {
            return false;
        }
    }
}

class Program
{
    static void Main()
    {
        Product product = new Product("Велосипед", 35000, 10);

        Console.WriteLine("Товар: " + product.Name);
        Console.WriteLine("Цена: " + product.Price);
        Console.WriteLine("Количество: " + product.Quantity);

        Console.WriteLine("Общая стоимость: " + product.GetTotalValue());

        if (product.Sell(3))
        {
            Console.WriteLine("Продажа успешна");
        }
        else
        {
            Console.WriteLine("Недостаточно товара");
        }

        Console.WriteLine("Осталось на складе: " + product.Quantity);

        Console.ReadKey();
    }
}
```
