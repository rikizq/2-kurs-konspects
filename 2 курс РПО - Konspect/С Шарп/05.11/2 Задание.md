```C#
using System;

class BankAccount
{
    private string accountNumber;
    private decimal balance;
        private string ownerName;

    public BankAccount(string number, string name, decimal money = 0)
    {
        accountNumber = number;
        ownerName = name;
        balance = money;
    }

    public string AccountNumber
    {
        get { return accountNumber; }
    }

    public string OwnerName
    {
        get { return ownerName; }
        set { ownerName = value; }
    }

    public decimal Balance
    {
        get { return balance; }
    }

    public void Deposit(decimal amount)
    {
        if (amount > 0)
        {
            balance = balance + amount;
            Console.WriteLine("Счет пополнен");
        }
        else
        {
            Console.WriteLine("Сумма должна быть больше нуля");
        }
    }

    public void Withdraw(decimal amount)
    {
        if (amount > 0 && amount <= balance)
        {
            balance = balance - amount;
            Console.WriteLine("Деньги сняты");
        }
        else
        {
            Console.WriteLine("Недостаточно средств или неверная сумма");
        }
    }

    public void DisplayInfo()
    {
        Console.WriteLine("Номер счета: " + accountNumber);
        Console.WriteLine("Владелец: " + ownerName);
        Console.WriteLine("Баланс: " + balance);
    }
}

class Program
{
    static void Main()
    {
        BankAccount account = new BankAccount("2200889675359", "Ярослав", 1000);

        account.DisplayInfo();

        account.Deposit(500);
        account.Withdraw(200);

        account.DisplayInfo();

        Console.ReadKey();
    }
```
