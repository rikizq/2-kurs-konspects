```C#
using System;

class Student
{
    private string fullName;
    private string group;
    private int[] grades;

    public Student(string fullName, string group, int[] grades)
    {
        this.fullName = fullName;
        this.group = group;
        this.grades = grades;
    }

    public string FullName
    {
        get { return fullName; }
    }

    public string Group
    {
        get { return group; }
    }

    public double GetAverageGrade()
    {
        int sum = 0;
        for (int i = 0; i < 5; i++)
        {
            sum = sum + grades[i];
        }

        double average = (double)sum / 5;
        return average;
    }

    public void DisplayInfo()
    {
        Console.WriteLine("Студент");
        Console.WriteLine("Имя: " + fullName);
        Console.WriteLine("Группа: " + group);
        Console.WriteLine("Средний балл: " + GetAverageGrade());
    }
}

class Program
{
    static void Main()
    {
        Student s = new Student("Владимир Путин", "9/2-РПО", new int[] { 5, 4, 3, 5, 4 });
        s.DisplayInfo();
    }
}
```
