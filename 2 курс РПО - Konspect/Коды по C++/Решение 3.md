#include <iostream>
#include <math.h>
#include <numbers>

using namespace std;

int main()
{
    setlocale(LC_ALL, "Russian");

    double x1, y1, x2, y2;

    cout << "Введите x1: ";
    cin >> x1;

    cout << "Введите y1: ";
    cin >> y1;

    cout << "Введите x2: ";
    cin >> x2;

    cout << "Введите y2: ";
    cin >> y2;

    double d = sqrt((x2 - x1) * (x2 - x1) + (y2 - y1) * (y2 - y1));

    d = round(d * 10000) / 10000;

    cout << "Расстояние: " << d << endl;
}
