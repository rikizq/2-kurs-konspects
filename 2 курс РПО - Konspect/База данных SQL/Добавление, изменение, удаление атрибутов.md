ADD Название_атрибута

Пример 1 Добавим в сущность Блюдо атрибут вес порции (г)
ALTER TABLE Блюдо
ADD Вес_порции INT NULL;

SELECT * FROM Блюдо

Пример 2 Добавим в ингредиент атрибут и калорийность с ограничением Check

ADD TABLE Ингредиент
ADD Калорийность SMALLINT
CHECK Калорийность > = 0;
SELECT * FROM Ингредиент

ADD TABLE Ингредиент
ADD Калорийность SMALLINT;
CHECK Ингредиент
ADD CONSTRAINT CH_Ингредиент_Калорийность
CHECK (Калорийность >= 0);

### Команда на изменение атрибутов: 
UPDATE Название_Сущности
SET Атрибут;

UPDATE Блюдо
SET Всего_ингредиентов = 1;

```
UPDATE Блюдо
SET Всего_ингрендиентов = (

	SELECT COUNT(*)
	FROM Состав
	WHERE ID_Блюда = Блюдо.ID 
)
```

При изменении и удалении атрибутов не должно быть противоречия с ограничениями, если есть противоречия, удалить ограничения, после этого можно удалять и изменять атрибуты. При необходимости снова добавить ограничения.

### Удаление атрибутов:
Удаляем ранее добавленные атрибуты
ALTER TABLE Блюдо
DROP COLUMN Вес_порции

ALTER TABLE Ингредиент
DROP CONSTRAINT ON Ингредиент_Калорийность
