Операторы запроса на вывод сущности:
1. FROM - в нем указываются какую сущность выводим (он обязателен)
2. SELECT - Выводит атрибуты и сущности (обязателен) SELECT * выводит все атрибуты
3. INSERT INTO - команда для заполнения сущности данными 
4. DEFAULT - устанавливает значения по умолчанию
5. WHERE - фильтрует строки по условию
