```C++
#include <iostream>
using namespace std;

int main()
{
    setlocale(LC_ALL, "Russian");
    int a, b, sum = 0;

    cout << "Введите два числа: ";
    cin >> a >> b;

    if (a > b)
    {
        int c = a;
        a = b;
        b = c;
    }

    for (int i = a; i <= b; i++)
        sum += i;

    cout << "Сумма чисел в диапазоне: " << sum;
}
```
