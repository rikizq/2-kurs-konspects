```C++
#include <iostream>
#include <clocale>
using namespace std;

int main()
{
    setlocale(LC_ALL, "Russian");

    int a, b;
    cout << "Введите число: ";
    cin >> a >> b;

    if (a > b)
        swap(a, b);

    cout << "Все числа: ";
    for (int i = a; i <= b; i++)
        cout << i << " ";

    cout << "\nЧетные: ";
    for (int i = a; i <= b; i++)
        if (i % 2 == 0)
            cout << i << " ";

    cout << "\nНечетные: ";
    for (int i = a; i <= b; i++)
        if (i % 2 != 0)
            cout << i << " ";

    cout << "\nКратные 7: ";
    for (int i = a; i <= b; i++)
        if (i % 7 == 0)
            cout << i << " ";
}
```
