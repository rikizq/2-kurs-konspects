#include <iostream>
using namespace std;

int main()
{
    setlocale(LC_ALL, "Russian");

    int a, b;
    cout << "Введите число: ";
    cin >> a >> b;

    if (a > b) 
    {
        int swp = a;
        a = b;
        b = swp;
    }

    cout << "Все числа: " << endl;
    for (int i = a; i <= b; i++)
        cout << i << endl;

    cout << "Четные: " << endl;
    for (int i = a; i <= b; i++)
        if (i % 2 == 0)
            cout << i << endl;

    cout << "Нечетные: " << endl;
    for (int i = a; i <= b; i++)
        if (i % 2 != 0)
            cout << i << endl;

    cout << "Кратные 7: " << endl;
    for (int i = a; i <= b; i++)
        if (i % 7 == 0)
            cout << i << endl;
}
