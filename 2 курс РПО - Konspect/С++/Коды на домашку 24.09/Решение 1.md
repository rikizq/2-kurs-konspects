``` C++
#include <iostream>
#include <math.h>
#include <numbers>

using namespace std;

int main()
{
    setlocale(LC_ALL, "Russian");

    double r;

    cout << "Введите радиус: ";
    cin >> r;

    double C = 2 * numbers::pi * r;
    double S = numbers::pi * r * r;

    C = round(C * 1000) / 1000;
    S = round(S * 1000) / 1000;

    cout << "Длина окружности: " << C << endl;
    cout << "Площадь круга: " << S << endl;

}
```
