```C++
#include <iostream>
#include <math.h>
#include <numbers>

using namespace std;

int main()
{
    setlocale(LC_ALL, "Russian");

    double r;

    cout << "Введите радиус: ";
    cin >> r;

    double V = 4.0 / 3.0 * numbers::pi * r * r * r;
    double S = 4 * numbers::pi * r * r;

    V = round(V * 100) / 100;
    S = round(S * 100) / 100;

    cout << "Объём: " << V << endl;
    cout << "Площадь поверхности: " << S << endl;
}
```
