```C++
#include <iostream>
#include <math.h>
#include <numbers>

using namespace std;

int main()
{
    setlocale(LC_ALL, "Russian");

    double a;

    cout << "Введите угол в градусах: ";
    cin >> a;

    double rad = a * numbers::pi / 180;

    double sin_a = sin(rad);
    double cos_a = cos(rad);

    double result = sin_a * sin_a + cos_a * cos_a;

    sin_a = round(sin_a * 1000) / 1000;
    cos_a = round(cos_a * 1000) / 1000;
    result = round(result * 1000) / 1000;

    cout << "sin(" << a << ") = " << sin_a << endl;
    cout << "cos(" << a << ") = " << cos_a << endl;
    cout << "sin^2 + cos^2 = " << result << endl;
}
```
