#include <iostream>
#include <math.h>
#include <numbers>

using namespace std;

int main()
{
    setlocale(LC_ALL, "Russian");

    double a, b, c;

    cout << "Введите сторону a: ";
    cin >> a;

    cout << "Введите сторону b: ";
    cin >> b;

    cout << "Введите сторону c: ";
    cin >> c;

    double p = (a + b + c) / 2;
    double S = sqrt(p * (p - a) * (p - b) * (p - c));

    p = round(p * 100.0) / 100.0;
    S = round(S * 100.0) / 100.0;

    cout << "Площадь треугольника: " << S << endl;

}
